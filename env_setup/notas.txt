# (1) Se houver um venv/conda ativo, saia dele
deactivate 2>/dev/null || conda deactivate 2>/dev/null || true

# (2) Desinstale versões antigas do RiskPilot — repita até pip não encontrar mais nada
pip uninstall -y riskpilot || true

# (3) Elimine diretórios residuais do pacote
python - <<'PY'
import site, shutil, pathlib, sys, importlib.util
for path in site.getsitepackages() + [site.getusersitepackages()]:
    for p in pathlib.Path(path).glob("riskpilot*"):
        print("removendo", p)
        shutil.rmtree(p, ignore_errors=True)
# idem para caches de byte-code
for p in pathlib.Path.home().rglob("__pycache__/riskpilot*"):
    shutil.rmtree(p, ignore_errors=True)
PY
